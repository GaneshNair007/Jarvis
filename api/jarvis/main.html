def cli_entrypoint():
    args = sys.argv[1:]
    if not args or args[0] == "serve":
        run_server()
    elif args[0] == "chat":
        if len(args) < 2:
            print("Error: Please provide a query string. Example: python -m jarvis.main chat 'System health'")
            sys.exit(1)
        run_cli_chat(args[1])
    elif args[0] == "health":
        run_health()
    elif args[0] == "audit":
        run_audit_verify()
    elif args[0] == "powers":
        run_powers_summary()
    elif args[0] == "voice":
        run_voice_mode()
    elif args[0] in ["--help", "-h", "help"]:
        print("""
JARVIS Operating System CLI
Usage:
  python -m jarvis.main serve       # Run production FastAPI & WebSocket server
  python -m jarvis.main chat "text" # Process a single query via terminal
  python -m jarvis.main health      # Print live host health metrics
  python -m jarvis.main audit       # Verify cryptographic audit chain integrity
  python -m jarvis.main powers      # List catalog power statistics
  python -m jarvis.main voice       # Launch interactive full-duplex voice console
""")
    else:
        print(f"Unknown command '{args[0]}'. Use: serve, chat, health, audit, powers, voice, --help")

def run_voice_mode():
    from jarvis.voice.wake import wake_detector
    from jarvis.voice.tts import tts_engine

    print("\n🎙️  JARVIS Full-Duplex Voice Mode Active")
    print("Wake word: 'Hey Jarvis' or press Enter for Push-to-Talk. Type 'exit' to quit.\n")
    tts_engine.speak_live("Jarvis voice subsystem online.", async_mode=True)
    wake_detector.start_listening()

    while True:
        try:
            cmd = input("voice> ")
            if cmd.strip().lower() in ["exit", "quit"]:
                wake_detector.stop_listening()
                print("Voice mode deactivated.")
                break
            elif not cmd.strip():
                wake_detector.trigger_push_to_talk("console_enter")
                print("Activated via Push-to-Talk.")
            else:
                run_cli_chat(cmd)
        except (KeyboardInterrupt, EOFError):
            wake_detector.stop_listening()
            break
