class AgentService:
    """Service layer managing dynamic agent lifecycles and roster allocation."""

    def __init__(self, uow_factory=UnitOfWork):
        self.uow_factory = uow_factory

    def spawn_worker_agent(
        self,
        agent_key: str,
        name: str,
        role: str,
        department_name: str,
        model_name: str = "gemini-2.5-pro",
        capabilities: Optional[List[str]] = None,
    ) -> AgentModel:
        """Dynamically instantiates and persists a specialized AI worker agent."""
        with self.uow_factory() as uow:
            # Check or resolve department
            dept = uow.session.query(uow.departments.model_class).filter_by(name=department_name).first()
            dept_id = dept.id if dept else None

            agent = uow.agents.register_agent(
                agent_key=agent_key,
                name=name,
                role=role,
                department_id=dept_id,
                model_name=model_name,
                capabilities=capabilities or [],
            )
            uow.commit()

            # Broadcast lifecycle event
            event_bus.publish(
                AgentLifecycleEvent(
                    agent_id=str(agent.id),
                    state="created",
                    reason=f"Dynamically spawned agent [{agent_key}] for department [{department_name}].",
                )
            )

            # Initialize live Redis state
            redis_state_store.set_agent_status(
                department=department_name,
                agent_name=agent_key,
                status="IDLE",
                current_task="initialized",
                progress=0,
            )

            logger.info(f"Dynamically spawned worker agent '{name}' ({agent.id}) on model '{model_name}'.")
            return agent

    def pause_agent(self, agent_id: uuid.UUID, reason: str = "Resource reclamation") -> Optional[AgentModel]:
        """Pauses worker agent execution and frees resources."""
        with self.uow_factory() as uow:
            agent = uow.agents.update_status(agent_id, "PAUSED", reason=reason)
            if agent:
                event_bus.publish(
                    AgentLifecycleEvent(
                        agent_id=str(agent.id),
                        state="paused",
                        reason=reason,
                    )
                )
            uow.commit()
            return agent

    def resume_agent(self, agent_id: uuid.UUID) -> Optional[AgentModel]:
        """Resumes a paused agent back to active duty."""
        with self.uow_factory() as uow:
            agent = uow.agents.update_status(agent_id, "ACTIVE", reason="Resumed to active fleet")
            if agent:
                event_bus.publish(
                    AgentLifecycleEvent(
                        agent_id=str(agent.id),
                        state="resumed",
                        reason="Resumed to active fleet",
                    )
                )
            uow.commit()
            return agent

    def retire_agent(self, agent_id: uuid.UUID, reason: str = "Goal achieved") -> Optional[AgentModel]:
        """Retires a worker agent, preserving full audit history while decommissioning."""
        with self.uow_factory() as uow:
            agent = uow.agents.update_status(agent_id, "RETIRED", reason=reason)
            if agent:
                event_bus.publish(
                    AgentLifecycleEvent(
                        agent_id=str(agent.id),
                        state="retired",
                        reason=reason,
                    )
                )
            uow.commit()
            return agent

    def get_active_workforce(self) -> List[AgentModel]:
        with self.uow_factory() as uow:
            return uow.agents.get_available_agents()

class GoalService:
    """Service layer orchestrating Goal lifecycle and DAG persistence."""

    def __init__(self, uow_factory=UnitOfWork):
        self.uow_factory = uow_factory

    def create_goal(
        self,
        title: str,
        description: str = "",
        priority: str = "HIGH",
        created_by: str = "ganesh",
        metadata: Optional[Dict[str, Any]] = None,
    ) -> GoalModel:
        with self.uow_factory() as uow:
            goal = uow.goals.create_goal(
                title=title,
                description=description,
                priority=priority,
                created_by=created_by,
                metadata=metadata,
            )
            uow.audits.record_audit(
                action="CREATE_GOAL",
                target=f"Goal:{goal.id}",
                metadata={"title": title, "priority": priority},
            )
            uow.commit()
            logger.info(f"Goal '{title}' ({goal.id}) successfully created and persisted.")
            return goal

    def store_task_graph_dag(
        self,
        goal_id: uuid.UUID,
        graph: TaskGraph,
    ) -> uuid.UUID:
        """
        Ingests in-memory TaskGraph DAG into relational database models:
        1. Creates TaskGraphModel
        2. Creates TaskModel for each node
        3. Creates TaskDependencyModel for each edge
        All executed atomically within a single UnitOfWork transaction.
        """
        with self.uow_factory() as uow:
            db_graph = uow.tasks.create_task_graph(
                goal_id=goal_id,
                graph_name=f"DAG_{graph.goal_id}",
                metadata={"total_nodes": len(graph.nodes)},
            )
            uow.flush()

            # Map in-memory task_id (str) -> database task_id (UUID)
            key_to_uuid: Dict[str, uuid.UUID] = {}

            # 1. Insert tasks
            for node in graph.nodes.values():
                db_task = uow.tasks.create_task(
                    task_graph_id=db_graph.id,
                    goal_id=goal_id,
                    task_key=node.task_id,
                    name=node.name,
                    description=node.description,
                    department=node.department,
                    priority=node.priority.value,
                    assigned_agent=node.assigned_agent,
                    tool_requirements=node.tool_requirements,
                    metadata=node.metadata,
                )
                uow.flush()
                key_to_uuid[node.task_id] = db_task.id

            # 2. Insert dependency edges
            for node in graph.nodes.values():
                child_uuid = key_to_uuid[node.task_id]
                for dep_key in node.dependencies:
                    if dep_key in key_to_uuid:
                        parent_uuid = key_to_uuid[dep_key]
                        uow.dependencies.add_dependency(
                            task_id=child_uuid,
                            depends_on_task_id=parent_uuid,
                        )

            uow.commit()
            logger.info(
                f"Persisted DAG '{graph.goal_id}' with {len(graph.nodes)} tasks into TaskGraph ({db_graph.id})."
            )
            return db_graph.id

    def complete_goal(self, goal_id: uuid.UUID) -> Optional[GoalModel]:
        with self.uow_factory() as uow:
            goal = uow.goals.update_status(goal_id, "COMPLETED")
            if goal:
                tasks = uow.tasks.session.query(uow.tasks.model_class).filter_by(goal_id=goal_id).all()
                event_bus.publish(
                    GoalCompletedEvent(
                        goal_id=str(goal.id),
                        title=goal.title,
                        total_tasks=len(tasks),
                    )
                )
            uow.commit()
            return goal

class MemoryService:
    def __init__(self, uow_factory=UnitOfWork):
        self.uow_factory = uow_factory

    def record_memory(
        self,
        category: str,
        content: Dict[str, Any],
        importance_score: float = 1.0,
        source_task_id: Optional[uuid.UUID] = None,
        goal_id: Optional[uuid.UUID] = None,
    ) -> MemoryModel:
        with self.uow_factory() as uow:
            mem = uow.memories.store_memory(
                category=category,
                content=content,
                importance_score=importance_score,
                source_task_id=source_task_id,
                goal_id=goal_id,
            )
            uow.commit()
            return mem

    def recall(
        self,
        query: str,
        category: Optional[str] = None,
        limit: int = 5,
    ) -> List[MemoryModel]:
        """Retrieves top semantic memories matching the query."""
        with self.uow_factory() as uow:
            return uow.memories.search_memory(term=query, category=category, limit=limit)

class OpportunityService:
    def __init__(self, uow_factory=UnitOfWork):
        self.uow_factory = uow_factory

    def ingest_opportunity(
        self,
        opportunity_type: str,
        title: str,
        source: str,
        score: float,
        justification: str = "",
        metadata: Optional[Dict[str, Any]] = None,
    ) -> OpportunityModel:
        """Stores new opportunity and broadcasts discovery event."""
        with self.uow_factory() as uow:
            opp = uow.opportunities.record_opportunity(
                opportunity_type=opportunity_type,
                title=title,
                source=source,
                score=score,
                justification=justification,
                metadata=metadata,
            )
            uow.commit()

            event_bus.publish(
                OpportunityDiscoveredEvent(
                    opportunity_id=str(opp.id),
                    opportunity_type=opp.opportunity_type,
                    title=opp.title,
                    score=opp.score,
                    source=opp.source,
                )
            )
            logger.info(f"Opportunity discovered: '{title}' [{opportunity_type}] (Score: {score})")
            return opp

    def get_top_hackathons(self, min_score: float = 0.75, limit: int = 10) -> List[OpportunityModel]:
        with self.uow_factory() as uow:
            return uow.opportunities.get_best_hackathons(min_score=min_score, limit=limit)

    def get_top_internships(self, limit: int = 10) -> List[OpportunityModel]:
        with self.uow_factory() as uow:
            return uow.opportunities.filter_by_type("internship", limit=limit)

class TaskService:
    """Service layer managing task execution states and live Redis status."""

    def __init__(self, uow_factory=UnitOfWork):
        self.uow_factory = uow_factory

    def get_ready_unblocked_tasks(self, task_graph_id: uuid.UUID) -> List[TaskModel]:
        """Queries database for all tasks whose dependencies are 100% satisfied."""
        with self.uow_factory() as uow:
            unblocked = uow.tasks.get_unblocked_tasks(task_graph_id)
            uow.commit()
            return unblocked

    def start_task(
        self,
        task_id: uuid.UUID,
        agent_name: str,
        department: str,
    ) -> Optional[TaskModel]:
        """Transitions task to RUNNING, records assignment, and updates Redis live status."""
        with self.uow_factory() as uow:
            task = uow.tasks.update_task_status(task_id, "RUNNING")
            if task:
                task.assigned_agent = agent_name
                # Find or register agent
                agent = uow.agents.get_agent_by_key(agent_name)
                if agent:
                    uow.assignments.assign_task(task_id=task.id, agent_id=agent.id)

                uow.commit()

                # Update live Redis state
                redis_state_store.set_agent_status(
                    department=department,
                    agent_name=agent_name,
                    status="RUNNING",
                    current_task=task.name,
                    progress=10,
                )
                logger.info(f"Task '{task.name}' ({task.id}) started by [{agent_name}].")
            return task

    def complete_task(
        self,
        task_id: uuid.UUID,
        outputs: Dict[str, Any],
        tool_name: str = "agent_executor",
        duration_seconds: float = 0.5,
    ) -> Optional[TaskModel]:
        """Transitions task to COMPLETED, records execution audit, publishes event, and clears Redis."""
        with self.uow_factory() as uow:
            task = uow.tasks.update_task_status(task_id, "COMPLETED", outputs=outputs)
            if task:
                # Find agent
                agent_id = None
                if task.assigned_agent:
                    ag = uow.agents.get_agent_by_key(task.assigned_agent)
                    if ag:
                        agent_id = ag.id

                # Record execution
                uow.executions.record_execution(
                    task_id=task.id,
                    tool_name=tool_name,
                    input_data=task.metadata_json or {},
                    output_data=outputs,
                    status="SUCCESS",
                    execution_time=duration_seconds,
                    agent_id=agent_id,
                )

                # Record memory if insights exist
                uow.memories.store_memory(
                    category=task.department,
                    content={"task_name": task.name, "outputs": outputs},
                    source_task_id=task.id,
                    goal_id=task.goal_id,
                )

                uow.commit()

                # Publish domain event
                event_bus.publish(
                    TaskCompletedEvent(
                        task_id=str(task.id),
                        task_name=task.name,
                        department=task.department,
                        agent_id=task.assigned_agent or "unassigned",
                        outputs=outputs,
                    )
                )

                # Update live Redis state
                if task.assigned_agent:
                    redis_state_store.set_agent_status(
                        department=task.department,
                        agent_name=task.assigned_agent,
                        status="IDLE",
                        current_task="completed",
                        progress=100,
                    )

                logger.info(f"Task '{task.name}' ({task.id}) marked COMPLETED.")
            return task

def complete_task(
    self,
    task_id: uuid.UUID,
    outputs: Dict[str, Any],
    tool_name: str = "agent_executor",
    duration_seconds: float = 0.5,
) -> Optional[TaskModel]:
    """Transitions task to COMPLETED, records execution audit, publishes event, and clears Redis."""
    with self.uow_factory() as uow:
        task = uow.tasks.update_task_status(task_id, "COMPLETED", outputs=outputs)
        if task:
            # Find agent
            agent_id = None
            if task.assigned_agent:
                ag = uow.agents.get_agent_by_key(task.assigned_agent)
                if ag:
                    agent_id = ag.id

            # Record execution
            uow.executions.record_execution(
                task_id=task.id,
                tool_name=tool_name,
                input_data=task.metadata_json or {},
                output_data=outputs,
                status="SUCCESS",
                execution_time=duration_seconds,
                agent_id=agent_id,
            )

            # Record memory if insights exist
            uow.memories.store_memory(
                category=task.department,
                content={"task_name": task.name, "outputs": outputs},
                source_task_id=task.id,
                goal_id=task.goal_id,
            )

            uow.commit()

            # Publish domain event
            event_bus.publish(
                TaskCompletedEvent(
                    task_id=str(task.id),
                    task_name=task.name,
                    department=task.department,
                    agent_id=task.assigned_agent or "unassigned",
                    outputs=outputs,
                )
            )

            # Update live Redis state
            if task.assigned_agent:
                redis_state_store.set_agent_status(
                    department=task.department,
                    agent_name=task.assigned_agent,
                    status="IDLE",
                    current_task="completed",
                    progress=100,
                )

            logger.info(f"Task '{task.name}' ({task.id}) marked COMPLETED.")
        return task

def start_task(
    self,
    task_id: uuid.UUID,
    agent_name: str,
    department: str,
) -> Optional[TaskModel]:
    """Transitions task to RUNNING, records assignment, and updates Redis live status."""
    with self.uow_factory() as uow:
        task = uow.tasks.update_task_status(task_id, "RUNNING")
        if task:
            task.assigned_agent = agent_name
            # Find or register agent
            agent = uow.agents.get_agent_by_key(agent_name)
            if agent:
                uow.assignments.assign_task(task_id=task.id, agent_id=agent.id)

            uow.commit()

            # Update live Redis state
            redis_state_store.set_agent_status(
                department=department,
                agent_name=agent_name,
                status="RUNNING",
                current_task=task.name,
                progress=10,
            )
            logger.info(f"Task '{task.name}' ({task.id}) started by [{agent_name}].")
        return task
