class ExecutionAnalyzer:
    """Analyzes runtime telemetry and execution logs of task graphs."""

    def analyze_graph(self, graph: TaskGraph) -> ExecutionAnalysis:
        total = len(graph.nodes)
        completed = 0
        failed = 0
        retried = 0
        bottlenecks = []
        repeated_failures: Dict[str, int] = {}

        for task_id, node in graph.nodes.items():
            if node.status == TaskStatus.COMPLETED:
                completed += 1
            elif node.status == TaskStatus.FAILED:
                failed += 1
                err = node.metadata.get("last_error", "Unknown error")
                repeated_failures[err] = repeated_failures.get(err, 0) + 1

            if node.retry_count > 0:
                retried += 1
                if node.retry_count >= 2:
                    bottlenecks.append(f"Task '{node.name}' ({task_id}) required {node.retry_count} retries.")

        rate = round((completed / max(total, 1)) * 100.0, 1)

        return ExecutionAnalysis(
            total_tasks=total,
            completed_tasks=completed,
            failed_tasks=failed,
            retried_tasks=retried,
            completion_rate_pct=rate,
            bottlenecks=bottlenecks,
            repeated_failures=repeated_failures,
        )

class GapDetector:
    """Inspects task failures and tool requirements to detect system capability gaps."""

    def __init__(self, known_capabilities: Set[str] = None):
        self.known_capabilities = known_capabilities or {
            "web_search",
            "ast_mapper",
            "git_cli",
            "data_filter",
            "doc_parser",
            "deep_reasoner",
            "decision_matrix",
            "roadmap_generator",
            "pytest",
            "dlp_scan",
            "system_health",
        }

    def detect_gaps(self, graph: TaskGraph) -> List[CapabilityGap]:
        gaps: List[CapabilityGap] = []
        missing_tools: Dict[str, List[str]] = {}

        for task_id, node in graph.nodes.items():
            # 1. Check for missing tool requirements
            for req_tool in node.tool_requirements:
                if req_tool not in self.known_capabilities:
                    if req_tool not in missing_tools:
                        missing_tools[req_tool] = []
                    missing_tools[req_tool].append(task_id)

            # 2. Check for task failure causes
            if node.status == TaskStatus.FAILED:
                err = node.metadata.get("last_error", "")
                if "timeout" in err.lower():
                    gaps.append(
                        CapabilityGap(
                            gap_type="TIMEOUT",
                            name=f"Execution Timeout in {node.name}",
                            description=f"Task '{node.name}' timed out repeatedly: {err}",
                            affected_tasks=[task_id],
                            suggested_capability=f"optimize_{node.department}_pipeline",
                            severity="HIGH",
                        )
                    )
                elif "unredacted" in err.lower() or "security" in err.lower():
                    gaps.append(
                        CapabilityGap(
                            gap_type="SECURITY_DLP_VIOLATION",
                            name=f"DLP Leak Prevention in {node.name}",
                            description=f"Task '{node.name}' failed DLP quality gate.",
                            affected_tasks=[task_id],
                            suggested_capability="enhanced_dlp_prefilter",
                            severity="CRITICAL",
                        )
                    )

        for missing_tool, task_ids in missing_tools.items():
            gaps.append(
                CapabilityGap(
                    gap_type="MISSING_TOOL",
                    name=f"Missing Tool: {missing_tool}",
                    description=f"Required tool '{missing_tool}' is not registered in system capability registry.",
                    affected_tasks=task_ids,
                    suggested_capability=f"tool_integration_{missing_tool}",
                    severity="HIGH" if len(task_ids) > 1 else "MEDIUM",
                )
            )

        return gaps

class ImprovementGenerator:
    """Generates structured Improvement proposals ready for Human Review in the Backlog."""

    def generate_from_gaps(self, gaps: List[CapabilityGap], analysis: ExecutionAnalysis) -> List[ImprovementProposal]:
        proposals: List[ImprovementProposal] = []

        for gap in gaps:
            priority = "CRITICAL" if gap.severity == "CRITICAL" else ("HIGH" if gap.severity == "HIGH" else "MEDIUM")
            prop = ImprovementProposal(
                title=f"Resolve Gap: {gap.name}",
                description=gap.description,
                category="capability",
                priority=priority,
                proposed_changes={
                    "gap_type": gap.gap_type,
                    "suggested_capability": gap.suggested_capability,
                    "affected_tasks": gap.affected_tasks,
                    "action_required": f"Implement and register '{gap.suggested_capability}' in CapabilityRegistry.",
                },
            )
            proposals.append(prop)

        # Bottlenecks
        for bottleneck in analysis.bottlenecks:
            prop = ImprovementProposal(
                title=f"Optimize Bottleneck: {bottleneck[:60]}",
                description=bottleneck,
                category="performance",
                priority="MEDIUM",
                proposed_changes={
                    "action_required": "Refine agent prompts, increase timeouts, or optimize tool calling."
                },
            )
            proposals.append(prop)

        return proposals

class OpportunityDetector:
    """Discovers expansion opportunities and reusable skills from task execution outputs."""

    def detect_opportunities(self, graph: TaskGraph) -> List[Opportunity]:
        opportunities: List[Opportunity] = []

        for task_id, node in graph.nodes.items():
            if node.status == TaskStatus.COMPLETED:
                for key, val in node.outputs.items():
                    key_lower = key.lower()
                    if any(term in key_lower for term in ["opportunity", "suggest", "idea", "product", "export"]):
                        opportunities.append(
                            Opportunity(
                                category="PRODUCT" if "product" in key_lower else "SKILL",
                                title=f"Opportunity from {node.name}",
                                description=str(val),
                                source_task=task_id,
                                impact_level="HIGH",
                            )
                        )

                    if "repo_created" in key_lower:
                        opportunities.append(
                            Opportunity(
                                category="ENGINEERING",
                                title=f"Open Source Asset Created",
                                description=f"Automate CI/CD and GSoC alignment for created repository: {val}",
                                source_task=task_id,
                                impact_level="HIGH",
                            )
                        )

        # Overall Graph Opportunity
        if graph.is_completed() and len(graph.nodes) >= 5:
            opportunities.append(
                Opportunity(
                    category="WORKFLOW",
                    title=f"Synthesize Reusable Workflow Template",
                    description=f"Goal '{graph.goal_id}' completed with 100% success; package DAG topology into standard skill template.",
                    source_task="goal_graph",
                    impact_level="TRANSFORMATIVE",
                )
            )

        return opportunities

def detect_opportunities(self, graph: TaskGraph) -> List[Opportunity]:
    opportunities: List[Opportunity] = []

    for task_id, node in graph.nodes.items():
        if node.status == TaskStatus.COMPLETED:
            for key, val in node.outputs.items():
                key_lower = key.lower()
                if any(term in key_lower for term in ["opportunity", "suggest", "idea", "product", "export"]):
                    opportunities.append(
                        Opportunity(
                            category="PRODUCT" if "product" in key_lower else "SKILL",
                            title=f"Opportunity from {node.name}",
                            description=str(val),
                            source_task=task_id,
                            impact_level="HIGH",
                        )
                    )

                if "repo_created" in key_lower:
                    opportunities.append(
                        Opportunity(
                            category="ENGINEERING",
                            title=f"Open Source Asset Created",
                            description=f"Automate CI/CD and GSoC alignment for created repository: {val}",
                            source_task=task_id,
                            impact_level="HIGH",
                        )
                    )

    # Overall Graph Opportunity
    if graph.is_completed() and len(graph.nodes) >= 5:
        opportunities.append(
            Opportunity(
                category="WORKFLOW",
                title=f"Synthesize Reusable Workflow Template",
                description=f"Goal '{graph.goal_id}' completed with 100% success; package DAG topology into standard skill template.",
                source_task="goal_graph",
                impact_level="TRANSFORMATIVE",
            )
        )

    return opportunities

class ReflectionEngine:
    """Unified engine coordinating all reflection sub-engines."""

    def __init__(self):
        self.analyzer = ExecutionAnalyzer()
        self.gap_detector = GapDetector()
        self.opportunity_detector = OpportunityDetector()
        self.improvement_generator = ImprovementGenerator()
        self.scorer = ReflectionScorer()

    def generate_reflection(self, graph: TaskGraph) -> UnifiedReflectionReport:
        analysis = self.analyzer.analyze_graph(graph)
        gaps = self.gap_detector.detect_gaps(graph)
        opportunities = self.opportunity_detector.detect_opportunities(graph)
        proposals = self.improvement_generator.generate_from_gaps(gaps, analysis)
        constitution_score = self.scorer.score_execution(graph, analysis)

        return UnifiedReflectionReport(
            goal_id=graph.goal_id,
            analysis=analysis,
            constitution_score=constitution_score,
            gaps=gaps,
            opportunities=opportunities,
            improvement_proposals=proposals,
        )

class ReflectionScorer:
    """Evaluates task execution against Constitution pillars and constraints."""

    def __init__(self, constitution_dir: str = "jarvis/constitution"):
        self.constitution_dir = Path(constitution_dir)
        self.constraints = self._load_yaml("constraints.yaml")
        self.values = self._load_yaml("values.yaml")

    def _load_yaml(self, filename: str) -> Dict[str, Any]:
        path = self.constitution_dir / filename
        if path.exists():
            try:
                return yaml.safe_load(path.read_text(encoding="utf-8")) or {}
            except Exception:
                return {}
        return {}

    def score_execution(self, graph: TaskGraph, analysis: ExecutionAnalysis) -> ConstitutionScore:
        violations: List[str] = []
        compliances: List[str] = []

        # 1. Constraint: Mandatory Verification Gate
        all_verified = True
        for tid, node in graph.nodes.items():
            if node.status == TaskStatus.COMPLETED:
                verif = node.metadata.get("verification")
                if not verif or not verif.get("passed"):
                    all_verified = False
                    violations.append(f"Task '{tid}' marked COMPLETED without verified QA gate.")

        if all_verified and analysis.completed_tasks > 0:
            compliances.append("100% of completed tasks passed multi-stage QA & DLP verification gates.")

        # 2. Value: Verifiable Truth (Reliability over Illusion)
        if analysis.completion_rate_pct >= 90.0:
            compliances.append(f"High verifiable completion rate ({analysis.completion_rate_pct}%).")
        elif analysis.failed_tasks > 0 and analysis.completed_tasks == 0:
            violations.append("Zero successful completions in goal execution.")

        # 3. Compute Scores
        constraints_score = 1.0 if not violations else max(0.0, 1.0 - (len(violations) * 0.3))
        values_score = round(min(1.0, analysis.completion_rate_pct / 100.0), 2)
        overall_score = round((constraints_score * 0.6) + (values_score * 0.4), 2)

        return ConstitutionScore(
            overall_score=overall_score,
            values_score=values_score,
            constraints_score=constraints_score,
            violations=violations,
            compliances=compliances,
        )
