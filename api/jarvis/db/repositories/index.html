class AgentRepository(BaseRepository[AgentModel]):
    def __init__(self, session: Session):
        super().__init__(session, AgentModel)

    def register_agent(
        self,
        agent_key: str,
        name: str,
        role: str,
        department_id: Optional[uuid.UUID] = None,
        model_name: str = "gemini-2.5-pro",
        capabilities: Optional[List[str]] = None,
        agent_id: Optional[uuid.UUID] = None,
    ) -> AgentModel:
        """Registers a worker agent and records initial CREATED lifecycle event."""
        existing = self.get_agent_by_key(agent_key)
        if existing:
            return existing

        aid = agent_id or uuid.uuid4()
        agent = AgentModel(
            id=aid,
            agent_key=agent_key.lower(),
            name=name,
            role=role,
            department_id=department_id,
            model_name=model_name,
            status="ACTIVE",
            capabilities=capabilities or [],
        )
        self.session.add(agent)

        # Record Lifecycle event
        self.record_lifecycle_event(
            agent_id=aid,
            state="created",
            reason="Worker agent registered into organizational roster.",
        )
        return agent

    def get_agent(self, agent_id: uuid.UUID) -> Optional[AgentModel]:
        return self.session.query(AgentModel).filter(AgentModel.id == agent_id).first()

    def get_agent_by_key(self, agent_key: str) -> Optional[AgentModel]:
        return self.session.query(AgentModel).filter(AgentModel.agent_key == agent_key.lower()).first()

    def get_department_agents(self, department_id: uuid.UUID) -> List[AgentModel]:
        return self.session.query(AgentModel).filter(AgentModel.department_id == department_id).all()

    def get_available_agents(self) -> List[AgentModel]:
        return self.session.query(AgentModel).filter(AgentModel.status == "ACTIVE").all()

    def update_status(self, agent_id: uuid.UUID, status: str, reason: str = "") -> Optional[AgentModel]:
        """Transitions agent state (e.g. ACTIVE -> PAUSED / RETIRED) and writes lifecycle record."""
        agent = self.get_agent(agent_id)
        if agent:
            prev_status = agent.status
            agent.status = status.upper()
            self.session.add(agent)

            lifecycle_state = status.lower()
            if lifecycle_state not in ("created", "paused", "resumed", "retired", "active"):
                lifecycle_state = "updated"

            self.record_lifecycle_event(
                agent_id=agent.id,
                state=lifecycle_state,
                reason=reason or f"State changed from {prev_status} to {agent.status}",
            )
        return agent

    def record_lifecycle_event(
        self,
        agent_id: uuid.UUID,
        state: str,
        reason: str = "",
    ) -> AgentLifecycleModel:
        """Appends an immutable lifecycle transition entry."""
        event = AgentLifecycleModel(
            id=uuid.uuid4(),
            agent_id=agent_id,
            state=state.lower(),
            reason=reason,
            timestamp=datetime.datetime.now(datetime.timezone.utc),
        )
        self.session.add(event)
        return event

    def get_lifecycle_history(self, agent_id: uuid.UUID) -> List[AgentLifecycleModel]:
        """Returns ordered timeline of lifecycle state transitions for an agent."""
        return (
            self.session.query(AgentLifecycleModel)
            .filter(AgentLifecycleModel.agent_id == agent_id)
            .order_by(AgentLifecycleModel.timestamp.asc())
            .all()
        )

class ArtifactRepository(BaseRepository[ArtifactModel]):
    def __init__(self, session: Session):
        super().__init__(session, ArtifactModel)

    def create_artifact(
        self,
        artifact_type: str,
        location: str,
        task_id: Optional[uuid.UUID] = None,
        checksum: str = "",
        metadata: Optional[Dict[str, Any]] = None,
        artifact_id: Optional[uuid.UUID] = None,
    ) -> ArtifactModel:
        """Records an artifact record, automatically computing SHA-256 checksum if file exists."""
        computed_checksum = checksum
        if not computed_checksum and Path(location).exists() and Path(location).is_file():
            try:
                content = Path(location).read_bytes()
                computed_checksum = hashlib.sha256(content).hexdigest()
            except Exception:
                pass

        art = ArtifactModel(
            id=artifact_id or uuid.uuid4(),
            task_id=task_id,
            artifact_type=artifact_type.lower(),
            location=location,
            checksum=computed_checksum,
            metadata_json=metadata or {},
            created_at=datetime.datetime.now(datetime.timezone.utc),
        )
        self.session.add(art)
        return art

    def get_artifact(self, artifact_id: uuid.UUID) -> Optional[ArtifactModel]:
        return self.session.query(ArtifactModel).filter(ArtifactModel.id == artifact_id).first()

    def get_task_artifacts(self, task_id: uuid.UUID) -> List[ArtifactModel]:
        return (
            self.session.query(ArtifactModel)
            .filter(ArtifactModel.task_id == task_id)
            .order_by(ArtifactModel.created_at.asc())
            .all()
        )

class AssignmentRepository(BaseRepository[AssignmentModel]):
    def __init__(self, session: Session):
        super().__init__(session, AssignmentModel)

    def assign_task(self, task_id: uuid.UUID, agent_id: uuid.UUID) -> AssignmentModel:
        """Creates an assignment record binding task to agent."""
        assignment = AssignmentModel(
            id=uuid.uuid4(),
            task_id=task_id,
            agent_id=agent_id,
            assigned_at=datetime.datetime.now(datetime.timezone.utc),
            completed_at=None,
        )
        self.session.add(assignment)
        return assignment

    def active_assignments(self) -> List[AssignmentModel]:
        """Returns all assignments currently pending or in-progress."""
        return self.session.query(AssignmentModel).filter(AssignmentModel.completed_at.is_(None)).all()

    def completed_assignments(self) -> List[AssignmentModel]:
        """Returns all completed assignments."""
        return self.session.query(AssignmentModel).filter(AssignmentModel.completed_at.isnot(None)).all()

    def complete_assignment(self, assignment_id: uuid.UUID) -> Optional[AssignmentModel]:
        """Marks assignment as completed with UTC timestamp."""
        assignment = self.session.query(AssignmentModel).filter(AssignmentModel.id == assignment_id).first()
        if assignment:
            assignment.completed_at = datetime.datetime.now(datetime.timezone.utc)
            self.session.add(assignment)
        return assignment

class AuditRepository(BaseRepository[AuditModel]):
    def __init__(self, session: Session):
        super().__init__(session, AuditModel)

    def record_audit(
        self,
        action: str,
        target: str,
        actor: str = "system",
        metadata: Optional[Dict[str, Any]] = None,

            query = query.filter(AuditModel.actor == actor)
        if action:
            query = query.filter(AuditModel.action == action)
        return query.order_by(AuditModel.timestamp.desc()).limit(limit).all()

class BaseRepository(Generic[T]):
    """Universal repository interface encapsulating transactional operations."""

    def __init__(self, session: Session, model_class: Optional[Type[T]] = None):
        self.session = session
        self.model_class = model_class

    def save(self, entity: T) -> T:
        """Adds entity to current session."""
        self.session.add(entity)
        return entity

    def delete(self, entity: T) -> None:
        """Marks entity for deletion."""
        self.session.delete(entity)

    def flush(self) -> None:
        """Flushes pending changes to the database without committing."""
        self.session.flush()

    def commit(self) -> None:
        """Commits transaction directly (preferred through UnitOfWork)."""
        self.session.commit()

    def get_by_id(self, entity_id: uuid.UUID) -> Optional[T]:
        """Generic primary key lookup."""
        if not self.model_class:
            raise NotImplementedError("model_class must be defined for generic get_by_id.")
        return self.session.get(self.model_class, entity_id)

    def list_all(self, limit: int = 100) -> List[T]:
        """Lists entities up to limit."""
        if not self.model_class:
            raise NotImplementedError("model_class must be defined for generic list_all.")
        return self.session.query(self.model_class).limit(limit).all()

class DependencyRepository(BaseRepository[TaskDependencyModel]):
    def __init__(self, session: Session):
        super().__init__(session, TaskDependencyModel)

    def add_dependency(self, task_id: uuid.UUID, depends_on_task_id: uuid.UUID) -> TaskDependencyModel:
        """Adds a directed dependency edge (task_id depends on depends_on_task_id)."""
        existing = (
            self.session.query(TaskDependencyModel)
            .filter(
                TaskDependencyModel.task_id == task_id,
                TaskDependencyModel.depends_on_task_id == depends_on_task_id,
            )
            .first()
        )
        if existing:
            return existing

        dep = TaskDependencyModel(
            id=uuid.uuid4(),
            task_id=task_id,
            depends_on_task_id=depends_on_task_id,
        )
        self.session.add(dep)
        return dep

    def get_dependencies(self, task_id: uuid.UUID) -> List[TaskDependencyModel]:
        """Returns all parent tasks that the given task depends on."""
        return self.session.query(TaskDependencyModel).filter(TaskDependencyModel.task_id == task_id).all()

        self,
        node_id: uuid.UUID,
        relationship: Optional[str] = None,
    ) -> List[KnowledgeNodeModel]:
        """Finds all adjacent target nodes connected by outgoing edges."""
        query = (
            self.session.query(KnowledgeNodeModel)
            .join(KnowledgeEdgeModel, KnowledgeEdgeModel.target_node_id == KnowledgeNodeModel.id)
            .filter(KnowledgeEdgeModel.source_node_id == node_id)
        )
        if relationship:
            query = query.filter(KnowledgeEdgeModel.relationship == relationship.upper())
        return query.all()

    def find_path(
        self,
        start_node_id: uuid.UUID,
        end_node_id: uuid.UUID,
        max_depth: int = 5,
    ) -> List[uuid.UUID]:
        """Breadth-First Search (BFS) finding shortest path between two knowledge nodes."""
        if start_node_id == end_node_id:
            return [start_node_id]

                    return path + [e.target_node_id]

                if e.target_node_id not in visited:
                    visited.add(e.target_node_id)
                    queue.append(path + [e.target_node_id])

        return []

    def __init__(self, session: Session):
        super().__init__(session, OpportunityModel)

    def record_opportunity(
        self,
        opportunity_type: str,
        title: str,
        source: str,
        score: float,
        justification: str = "",
        metadata: Optional[Dict[str, Any]] = None,
        opportunity_id: Optional[uuid.UUID] = None,
    ) -> OpportunityModel:
        """Stores a high-value opportunity found during web scout tasks."""
        opp = OpportunityModel(
            id=opportunity_id or uuid.uuid4(),
            opportunity_type=opportunity_type.lower(),
            source=source,
            title=title,
            score=max(0.0, min(1.0, score)),
            justification=justification,
            metadata_json=metadata or {},
            discovered_at=datetime.datetime.now(datetime.timezone.utc),
        )
        self.session.add(opp)
        return opp

        """Returns opportunities ranked by highest relevance score."""
        return self.session.query(OpportunityModel).order_by(OpportunityModel.score.desc()).limit(limit).all()

    def get_recent_opportunities(self, limit: int = 10) -> List[OpportunityModel]:
        """Returns newest opportunities discovered."""
        return self.session.query(OpportunityModel).order_by(OpportunityModel.discovered_at.desc()).limit(limit).all()

                OpportunityModel.opportunity_type == "hackathon",
                OpportunityModel.score >= min_score,
            )
            .order_by(OpportunityModel.score.desc())
            .limit(limit)
            .all()
        )

    def filter_by_type(self, opportunity_type: str, limit: int = 20) -> List[OpportunityModel]:
        return (
            self.session.query(OpportunityModel)

            .limit(limit)
            .all()

        return self.session.query(TaskGraphModel).filter(TaskGraphModel.id == graph_id).first()

    def create_task(
        self,
        task_graph_id: uuid.UUID,
        goal_id: uuid.UUID,
        task_key: str,
        name: str,
        description: str,
        department: str,
        priority: str = "MEDIUM",
        assigned_agent: Optional[str] = None,
        tool_requirements: Optional[List[str]] = None,
        metadata: Optional[Dict[str, Any]] = None,
        task_id: Optional[uuid.UUID] = None,
    ) -> TaskModel:
        """Creates a single TaskModel in the DAG."""
        task = TaskModel(
            id=task_id or uuid.uuid4(),
            task_graph_id=task_graph_id,
            goal_id=goal_id,
            task_key=task_key,
            name=name,
            description=description,
            department=department.lower(),
            priority=priority.upper(),
            status="PENDING",
            assigned_agent=assigned_agent,
            tool_requirements=tool_requirements or [],
            outputs={},
            metadata_json=metadata or {},
        )
        self.session.add(task)
        return task

    def get_task(self, task_id: uuid.UUID) -> Optional[TaskModel]:
        return self.session.query(TaskModel).filter(TaskModel.id == task_id).first()

    def get_task_by_key(self, task_graph_id: uuid.UUID, task_key: str) -> Optional[TaskModel]:
        return (
            self.session.query(TaskModel)
            .filter(TaskModel.task_graph_id == task_graph_id, TaskModel.task_key == task_key)
            .first()
        )

    def get_tasks_by_graph(self, task_graph_id: uuid.UUID) -> List[TaskModel]:
        return self.session.query(TaskModel).filter(TaskModel.task_graph_id == task_graph_id).all()

    def get_ready_tasks(self, task_graph_id: Optional[uuid.UUID] = None) -> List[TaskModel]:
        query = self.session.query(TaskModel).filter(TaskModel.status == "READY")
        if task_graph_id:
            query = query.filter(TaskModel.task_graph_id == task_graph_id)
        return query.all()

    def get_running_tasks(self, task_graph_id: Optional[uuid.UUID] = None) -> List[TaskModel]:
        query = self.session.query(TaskModel).filter(TaskModel.status == "RUNNING")
        if task_graph_id:
            query = query.filter(TaskModel.task_graph_id == task_graph_id)
        return query.all()

    def update_task_status(
        self,
        task_id: uuid.UUID,
        status: str,
        outputs: Optional[Dict[str, Any]] = None,
    ) -> Optional[TaskModel]:
        task = self.get_task(task_id)
        if task:
            task.status = status.upper()
            if outputs:
                new_outputs = dict(task.outputs)
                new_outputs.update(outputs)
                task.outputs = new_outputs
            if status.upper() == "COMPLETED":
                task.completed_at = datetime.datetime.now(datetime.timezone.utc)
            self.session.add(task)
        return task

    def get_unblocked_tasks(self, task_graph_id: uuid.UUID) -> List[TaskModel]:
        """
        Special DAG Query:
        Inspects all PENDING, READY, or BLOCKED tasks in the task graph.
        For each task, evaluates all required dependencies:
          - If 100% of dependencies are COMPLETED -> sets/confirms status READY and includes in output.
          - If any dependency is not COMPLETED -> marks BLOCKED.
        """
        candidate_tasks = (
            self.session.query(TaskModel)
            .filter(
                TaskModel.task_graph_id == task_graph_id,
                TaskModel.status.in_(["PENDING", "READY", "BLOCKED"]),
            )
            .all()
        )

        unblocked: List[TaskModel] = []
        for task in candidate_tasks:
            # Query dependencies of this task
            deps = self.session.query(TaskDependencyModel).filter(TaskDependencyModel.task_id == task.id).all()

            if not deps:
                # No dependencies -> immediately unblocked
                if task.status != "READY":
                    task.status = "READY"
                    self.session.add(task)
                unblocked.append(task)
            else:
                # Check status of every dependency
                all_done = True
                for d in deps:
                    parent_task = self.session.query(TaskModel).filter(TaskModel.id == d.depends_on_task_id).first()
                    if not parent_task or parent_task.status != "COMPLETED":
                        all_done = False
                        break

                if all_done:
                    if task.status != "READY":
                        task.status = "READY"
                        self.session.add(task)
                    unblocked.append(task)
                else:
                    if task.status != "BLOCKED":
                        task.status = "BLOCKED"
                        self.session.add(task)

        return unblocked

def get_unblocked_tasks(self, task_graph_id: uuid.UUID) -> List[TaskModel]:
    """
    Special DAG Query:
    Inspects all PENDING, READY, or BLOCKED tasks in the task graph.
    For each task, evaluates all required dependencies:
      - If 100% of dependencies are COMPLETED -> sets/confirms status READY and includes in output.
      - If any dependency is not COMPLETED -> marks BLOCKED.
    """
    candidate_tasks = (
        self.session.query(TaskModel)
        .filter(
            TaskModel.task_graph_id == task_graph_id,
            TaskModel.status.in_(["PENDING", "READY", "BLOCKED"]),
        )
        .all()
    )

    unblocked: List[TaskModel] = []
    for task in candidate_tasks:
        # Query dependencies of this task
        deps = self.session.query(TaskDependencyModel).filter(TaskDependencyModel.task_id == task.id).all()

        if not deps:
            # No dependencies -> immediately unblocked
            if task.status != "READY":
                task.status = "READY"
                self.session.add(task)
            unblocked.append(task)
        else:
            # Check status of every dependency
            all_done = True
            for d in deps:
                parent_task = self.session.query(TaskModel).filter(TaskModel.id == d.depends_on_task_id).first()
                if not parent_task or parent_task.status != "COMPLETED":
                    all_done = False
                    break

            if all_done:
                if task.status != "READY":
                    task.status = "READY"
                    self.session.add(task)
                unblocked.append(task)
            else:
                if task.status != "BLOCKED":
                    task.status = "BLOCKED"
                    self.session.add(task)

    return unblocked
