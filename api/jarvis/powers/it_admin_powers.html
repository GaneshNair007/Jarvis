    @staticmethod
    def disk_space_analysis(path: str = settings.disk_mount_path) -> Dict[str, Any]:
        """PWR-103: Disk space forecasting and directory usage breakdown."""
        usage = shutil.disk_usage(path)
        free_pct = round((usage.free / usage.total) * 100, 2)
        burn_rate_gb_day = 0.5  # Estimated baseline burn rate
        days_until_exhaustion = round((usage.free / (1024**3)) / max(burn_rate_gb_day, 0.01), 1)

        return {
            "power_id": "PWR-103",
            "power_name": "Disk-space forecast",
            "path": path,
            "total_gb": round(usage.total / (1024**3), 2),
            "used_gb": round(usage.used / (1024**3), 2),
            "free_gb": round(usage.free / (1024**3), 2),
            "free_percent": free_pct,
            "estimated_days_remaining": days_until_exhaustion,
            "status": "critical" if free_pct < 10 else ("warning" if free_pct < 20 else "healthy"),
        }

    @staticmethod
    def process_analysis(limit: int = 10) -> Dict[str, Any]:
        """PWR-105: Process analysis and CPU/RAM hog detection."""
        processes = []
        for p in psutil.process_iter(["pid", "name", "cpu_percent", "memory_percent", "status"]):
            try:
                processes.append(p.info)
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                continue

        top_by_cpu = sorted(processes, key=lambda x: x.get("cpu_percent") or 0, reverse=True)[:limit]
        top_by_ram = sorted(processes, key=lambda x: x.get("memory_percent") or 0, reverse=True)[:limit]

        return {
            "power_id": "PWR-105",
            "power_name": "Process analysis",
            "total_running_processes": len(processes),
            "top_cpu_consumers": top_by_cpu,
            "top_ram_consumers": top_by_ram,
        }

    @staticmethod
    def safe_process_termination(pid: int, force: bool = False) -> Dict[str, Any]:
        """PWR-106: Safely terminates a runaway process with safety verification."""
        try:
            p = psutil.Process(pid)
            name = p.name()
            # Protect system and self
            if pid <= 4 or "system" in name.lower() or pid == psutil.Process().pid:
                return {
                    "power_id": "PWR-106",
                    "success": False,
                    "error": f"Cannot terminate protected critical process {name} (PID: {pid})",
                }
            if force:
                p.kill()
            else:
                p.terminate()
            return {
                "power_id": "PWR-106",
                "success": True,
                "pid": pid,
                "name": name,
                "action": "SIGKILL" if force else "SIGTERM",
            }
        except psutil.NoSuchProcess:
            return {"power_id": "PWR-106", "success": False, "error": f"Process {pid} does not exist"}
        except Exception as e:
            return {"power_id": "PWR-106", "success": False, "error": str(e)}

    @staticmethod
    def check_ssl_certificate(hostname: str, port: int = 443) -> Dict[str, Any]:
        """PWR-121: Checks SSL/TLS certificate validity and expiration for a domain."""
        context = ssl.create_default_context()
        try:
            with socket.create_connection((hostname, port), timeout=5.0) as sock:
                with context.wrap_socket(sock, server_hostname=hostname) as ssock:
                    cert = ssock.getpeercert()
                    not_after_str = cert["notAfter"]
                    # e.g., 'May 15 12:00:00 2027 GMT'
                    expiry = datetime.datetime.strptime(not_after_str, "%b %d %H:%M:%S %Y %Z")
                    days_left = (expiry - datetime.datetime.now(datetime.timezone.utc).replace(tzinfo=None)).days

                    return {
                        "power_id": "PWR-121",
                        "power_name": "SSL certificate monitoring",
                        "hostname": hostname,
                        "status": "valid" if days_left > 14 else "expiring_soon",
                        "expires_at": expiry.strftime("%Y-%m-%d"),
                        "days_remaining": days_left,
                        "issuer": dict(x[0] for x in cert.get("issuer", [])),
                    }
        except Exception as e:
            return {
                "power_id": "PWR-121",
                "power_name": "SSL certificate monitoring",
                "hostname": hostname,
                "status": "error",
                "message": f"Failed to verify SSL certificate: {str(e)}",

    @staticmethod
    def docker_status() -> Dict[str, Any]:
        """PWR-100: Lists Docker containers with status, CPU, and memory usage."""
        try:
            ps_result = subprocess.run(
                ["docker", "ps", "-a", "--format", "{{.ID}}\t{{.Names}}\t{{.Status}}\t{{.Image}}"],
                capture_output=True,
                text=True,
                timeout=15,
            )
            if ps_result.returncode != 0:
                return {
                    "power_id": "PWR-100",
                    "status": "error",
                    "message": ps_result.stderr.strip() or "Docker not available",
                }

            containers: List[Dict[str, str]] = []
            for line in ps_result.stdout.strip().splitlines():
                parts = line.split("\t")
                if len(parts) >= 4:
                    containers.append(
                        {
                            "id": parts[0],
                            "name": parts[1],
                            "status": parts[2],
                            "image": parts[3],
                        }
                    )

            # Gather stats for running containers
            stats_result = subprocess.run(
                ["docker", "stats", "--no-stream", "--format", "{{.Container}}\t{{.CPUPerc}}\t{{.MemUsage}}"],
                capture_output=True,
                text=True,
                timeout=15,
            )
            stats_map: Dict[str, Dict[str, str]] = {}
            if stats_result.returncode == 0:
                for line in stats_result.stdout.strip().splitlines():
                    parts = line.split("\t")
                    if len(parts) >= 3:
                        stats_map[parts[0]] = {"cpu": parts[1], "memory": parts[2]}

            for c in containers:
                s = stats_map.get(c["id"], stats_map.get(c["name"], {}))
                c["cpu_percent"] = s.get("cpu", "N/A")
                c["memory_usage"] = s.get("memory", "N/A")

            return {
                "power_id": "PWR-100",
                "status": "success",
                "container_count": len(containers),
                "containers": containers,
            }
        except FileNotFoundError:
            return {"power_id": "PWR-100", "status": "error", "message": "Docker CLI not found on PATH"}
        except Exception as e:
            return {"power_id": "PWR-100", "status": "error", "message": str(e)}

    @staticmethod
    def service_status(service_name: Optional[str] = None) -> Dict[str, Any]:
        """PWR-105: Query Windows service status via sc query."""
        try:
            cmd = ["sc", "query"]
            if service_name:
                cmd.append(service_name)
            res = subprocess.run(cmd, capture_output=True, text=True, timeout=15)
            if res.returncode != 0:
                return {"power_id": "PWR-105", "status": "error", "message": res.stderr.strip() or res.stdout.strip()}

            services: List[Dict[str, str]] = []
            current: Dict[str, str] = {}
            for line in res.stdout.splitlines():
                line = line.strip()
                if line.startswith("SERVICE_NAME:"):
                    if current:
                        services.append(current)
                    current = {"service_name": line.split(":", 1)[1].strip()}
                elif line.startswith("DISPLAY_NAME:"):
                    current["display_name"] = line.split(":", 1)[1].strip()
                elif line.startswith("STATE"):
                    # e.g. STATE              : 4  RUNNING
                    state_part = line.split(":", 1)[1].strip()
                    current["state"] = state_part
            if current:
                services.append(current)

            return {
                "power_id": "PWR-105",
                "status": "success",
                "service_count": len(services),
                "services": services[:100],  # cap output
            }
        except Exception as e:
            return {"power_id": "PWR-105", "status": "error", "message": str(e)}

    @staticmethod
    def disk_cleanup_preview() -> Dict[str, Any]:
        """PWR-112: Preview temp/cache files that could be cleaned."""
        try:
            targets = [
                tempfile.gettempdir(),
                os.path.join(os.environ.get("LOCALAPPDATA", ""), "Temp"),
                os.path.join(os.environ.get("WINDIR", r"C:\Windows"), "Temp"),
            ]
            cleanup_items: List[Dict[str, Any]] = []
            total_size = 0

            for target_dir in targets:
                if not os.path.isdir(target_dir):
                    continue
                dir_size = 0
                file_count = 0
                try:
                    for root, dirs, files in os.walk(target_dir):
                        for f in files:
                            try:
                                fp = os.path.join(root, f)
                                dir_size += os.path.getsize(fp)
                                file_count += 1
                            except (OSError, PermissionError):
                                pass
                except (OSError, PermissionError):
                    pass
                total_size += dir_size
                cleanup_items.append(
                    {
                        "directory": target_dir,
                        "file_count": file_count,
                        "size_mb": round(dir_size / (1024**2), 2),
                    }
                )

            return {
                "power_id": "PWR-112",
                "status": "success",
                "total_reclaimable_mb": round(total_size / (1024**2), 2),
                "locations": cleanup_items,
            }
        except Exception as e:
            return {"power_id": "PWR-112", "status": "error", "message": str(e)}

    @staticmethod
    def kill_process(pid: int) -> Dict[str, Any]:
        """PWR-116: Kill a process by PID. Requires user approval."""
        try:
            proc = psutil.Process(pid)
            proc_name = proc.name()
            proc.terminate()
            gone, alive = psutil.wait_procs([proc], timeout=5)
            if alive:
                for p in alive:
                    p.kill()
            return {
                "power_id": "PWR-116",
                "status": "success",
                "pid": pid,
                "name": proc_name,
                "message": f"Process {proc_name} (PID {pid}) terminated",
                "approval_required": True,
            }
        except psutil.NoSuchProcess:
            return {
                "power_id": "PWR-116",
                "status": "error",
                "message": f"No process with PID {pid}",
                "approval_required": True,
            }
        except psutil.AccessDenied:
            return {
                "power_id": "PWR-116",
                "status": "error",
                "message": f"Access denied for PID {pid}",
                "approval_required": True,
            }
        except Exception as e:
            return {"power_id": "PWR-116", "status": "error", "message": str(e), "approval_required": True}

                ["powershell", "-NoProfile", "-Command", ps_script],

                if isinstance(events, dict):

                events = [{"raw": res.stdout[:3000]}]

            return {
                "power_id": "PWR-118",
                "status": "success",
                "keyword": keyword,
                "source": source,
                "event_count": len(events),
                "events": events[:count],

            return {"power_id": "PWR-118", "status": "error", "message": str(e)}

    @staticmethod
    def startup_programs() -> Dict[str, Any]:
        """PWR-119: List startup programs from common registry locations."""
        try:
            programs: List[Dict[str, str]] = []
            reg_paths = [
                (winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run"),
                (winreg.HKEY_LOCAL_MACHINE, r"Software\Microsoft\Windows\CurrentVersion\Run"),
            ]
            for hive, path in reg_paths:
                try:
                    key = winreg.OpenKey(hive, path)
                    i = 0
                    while True:
                        try:
                            name, value, _ = winreg.EnumValue(key, i)
                            programs.append(
                                {
                                    "name": name,
                                    "command": value,
                                    "location": f"{'HKCU' if hive == winreg.HKEY_CURRENT_USER else 'HKLM'}\\{path}",
                                }
                            )
                            i += 1
                        except OSError:
                            break
                    winreg.CloseKey(key)
                except OSError:
                    pass

            # Also check startup folder
            startup_folder = os.path.join(
                os.environ.get("APPDATA", ""),
                r"Microsoft\Windows\Start Menu\Programs\Startup",
            )
            if os.path.isdir(startup_folder):
                for f in os.listdir(startup_folder):
                    programs.append(
                        {"name": f, "command": os.path.join(startup_folder, f), "location": "Startup Folder"}
                    )

            return {
                "power_id": "PWR-119",
                "status": "success",
                "count": len(programs),
                "programs": programs,
            }
        except Exception as e:
            return {"power_id": "PWR-119", "status": "error", "message": str(e)}

                pass

            "status": "error",

            "approval_required": True,

    try:
        p = psutil.Process(pid)

        if pid <= 4 or "system" in name.lower() or pid == psutil.Process().pid:
            return {

        if force:
            p.kill()
        else:
            p.terminate()
        return {
            "power_id": "PWR-106",
            "success": True,
            "pid": pid,
            "name": name,
            "action": "SIGKILL" if force else "SIGTERM",
        }
    except psutil.NoSuchProcess:
        return {"power_id": "PWR-106", "success": False, "error": f"Process {pid} does not exist"}
    except Exception as e:
        return {"power_id": "PWR-106", "success": False, "error": str(e)}

                current["state"] = state_part
        if current:
            services.append(current)

        return {
            "power_id": "PWR-105",
            "status": "success",
            "service_count": len(services),
            "services": services[:100],  # cap output
        }
